```typescript
// Standard hook structure with proper typing
export function useGame(): UseGameReturn {
  // 1. useReducer for complex state management
  const [gameState, dispatch] = useReducer(gameReducer, initialGameState);

  // 2. Local state for loading and UI feedback
  const [isLoading, setIsLoading] = useState(false);

  // 3. Effects for persistence and side effects
  useEffect(() => {
    loadPersistedGame();
  }, []);

  // 4. Action creators with proper error handling
  const makeMove = useCallback(
    (direction: Direction) => {
      // Implementation with validation and error handling
    },
    [gameState, dispatch]
  );

  // 5. Return object with state and actions

}
```

- Implement proper dependency arrays for all hooks
- Avoid creating new objects in render path

### Error Handling and Validation

**Move Validation Requirements:**

```typescript
const validateMove = (gameState: GameState, direction: Direction): boolean => {
  // Prevent moves during animations
  if (gameState.isAnimating) return false;

  // Prevent moves when game is over
  if (gameState.gameStatus === GameStatus.LOST) return false;

  // Check if move would actually change board state
  return gameEngine.canMove(gameState.board, direction);
};
```

**Error Handling Patterns:**

- All async operations wrapped in try-catch blocks
- Graceful degradation for persistence failures
- User-friendly error states returned from hook
- Never throw errors from hook - always return error states
- Comprehensive logging for debugging support
