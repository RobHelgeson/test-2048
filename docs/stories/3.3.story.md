```typescript
// Value-based color mapping from theme system
const getTileColors = (value: number, themeColors: ThemeColors) => {
  const colorMap = {
    2: themeColors.tile2,
    4: themeColors.tile4,
    8: themeColors.tile8,
    16: themeColors.tile16,
    32: themeColors.tile32,
    64: themeColors.tile64,
    128: themeColors.tile128,
    256: themeColors.tile256,
    512: themeColors.tile512,
    1024: themeColors.tile1024,
    2048: themeColors.tile2048, // Special victory color
    // Higher values use highest tier color
  };

  return colorMap[value] || themeColors.tile2048;
};

// Typography scaling based on value and tile size
const calculateFontSize = (value: number, tileSize: number): number => {
  const baseSize = tileSize * 0.35; // 35% of tile size

  if (value < 100) return baseSize;
  if (value < 1000) return baseSize * 0.9;
  if (value < 10000) return baseSize * 0.8;
  return baseSize * 0.7;
};
```

### Responsive Tile Sizing Requirements

[Source: architecture/coding-standards.md#platform-checks]

**Platform-Specific Tile Sizing:**

```typescript
import { Platform, Dimensions } from 'react-native';

// Calculate tile size based on board dimensions
const calculateTileSize = (boardSize: number): number => {
  const spacing = 8; // 8pt grid spacing
  const tilesPerRow = 4;
  const totalSpacing = spacing * (tilesPerRow + 1); // Border + between tiles

  return (boardSize - totalSpacing) / tilesPerRow;
};

// Ensure minimum touch targets per platform
const MINIMUM_TOUCH_TARGETS = {
  ios: 44, // 44x44pt minimum
  android: 48, // 48x48dp minimum
  web: 32, // 32x32px minimum
};

const ensureMinimumTouchTarget = (calculatedSize: number): number => {
  const minimum = MINIMUM_TOUCH_TARGETS[Platform.OS] || 32;
  return Math.max(calculatedSize, minimum);
};
```

### Accessibility Implementation for Tiles

[Source: architecture/components.md#theme-system-component]

**Tile Accessibility Features:**

```typescript
// Accessibility configuration for tiles
const getTileAccessibility = (tile: Tile) => ({
  accessibilityRole: 'button' as const,
  accessibilityLabel: `Tile with value ${tile.value}`,
  accessibilityHint: `Located at row ${tile.row + 1}, column ${tile.col + 1}`,
  accessibilityState: {
    disabled: false,
    selected: false,
  },
});

// High contrast text colors for accessibility
const getAccessibleTextColor = (
  backgroundColor: string,
  themeColors: ThemeColors
): string => {
  // Calculate contrast ratio and return appropriate text color
  // Use light text on dark backgrounds, dark text on light backgrounds
  return isLightColor(backgroundColor)
    ? themeColors.textPrimary
    : themeColors.textInverse;
};
```

### Design System Integration for Tiles

[Source: architecture/frontend-architecture.md#component-template]

**Tile Styling with Theme Integration:**

```typescript
// 8pt grid system spacing for tile styling
const TILE_SPACING = {
  border: 2,
  borderRadius: 4,
  shadowOffset: 2,
  textPadding: 4,
};

// Create tile styles with theme integration
const createTileStyles = (colors: ThemeColors, value: number, size: number) => {
  const tileColor = getTileColors(value, colors);
  const textColor = getAccessibleTextColor(tileColor, colors);
  const fontSize = calculateFontSize(value, size);

  return StyleSheet.create({
    container: {
      width: size,
      height: size,
      backgroundColor: tileColor,
      borderRadius: TILE_SPACING.borderRadius,
      justifyContent: 'center',
      alignItems: 'center',
      ...Platform.select({
        ios: {
          shadowColor: colors.shadow,
          shadowOffset: { width: 0, height: TILE_SPACING.shadowOffset },
          shadowOpacity: 0.2,
          shadowRadius: 2,
        },
        android: {
          elevation: 2,
        },
        web: {
          boxShadow: `0px ${TILE_SPACING.shadowOffset}px 4px ${colors.shadow}20`,
        },
      }),
    } as ViewStyle,
    text: {
      color: textColor,
      fontSize: fontSize,
      fontWeight: '700' as const,
      textAlign: 'center' as const,
    } as TextStyle,
    victoryTile: {
      // Special styling for 2048 tile
      backgroundColor: colors.victory,
      borderWidth: 2,
      borderColor: colors.victoryBorder,
    } as ViewStyle,
  });
};
```
