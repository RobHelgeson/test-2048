**Component Structure Pattern for GameHeader:**

```typescript
import React from 'react';
import { StyleSheet, ViewStyle, TextStyle } from 'react-native';
import { useTheme } from '@/hooks/useTheme';
import { useGameStore } from '@/stores/gameStore';
import { ThemedView } from '@/components/themed/ThemedView';
import { ThemedText } from '@/components/themed/ThemedText';
import { Button } from '@/components/ui/Button';

interface GameHeaderProps {
  style?: ViewStyle;
  onNewGame?: () => void;
}

export function GameHeader({ style, onNewGame }: GameHeaderProps) {
  const theme = useTheme();
  const { score, bestScore, gameStatus, newGame } = useGameStore();
  const dynamicStyles = createHeaderStyles(theme.colors);

  const handleNewGame = () => {
    newGame();
    onNewGame?.();
  };

  return (
    <ThemedView style={[dynamicStyles.container, style]}>
      <ThemedView style={dynamicStyles.scoresContainer}>
        <ScoreDisplay label="Score" value={score} />
        <ScoreDisplay label="Best" value={bestScore} />
      </ThemedView>
      <GameStatusIndicator status={gameStatus} />
      <Button title="New Game" onPress={handleNewGame} variant="primary" />
    </ThemedView>
  );
}
```

// Score formatting utility
const formatScore = (score: number): string => {
  if (score < 1000) return score.toString();
  if (score < 1000000) return `${(score / 1000).toFixed(1)}K`;
  return `${(score / 1000000).toFixed(1)}M`;
};

// Status message mapping
const getStatusMessage = (status: GameStatus): string => {
  switch (status) {
    case 'playing':
      return 'Playing';
    case 'won':
      return 'You Won!';
    case 'lost':
      return 'Game Over';
    default:
      return 'Playing';
  }
};

// Status color mapping
const getStatusColor = (status: GameStatus, colors: ThemeColors): string => {
  switch (status) {
    case 'playing':
      return colors.accent;
    case 'won':
      return colors.victory;
    case 'lost':
      return colors.danger;
    default:
      return colors.accent;
  }
};

### Animation Requirements for Score Updates

[Source: architecture/tech-stack.md]

**Score Animation Implementation:**

```typescript
import Animated, { useSharedValue, withSpring, withSequence, runOnJS } from 'react-native-reanimated';

// Score change animation
const scoreAnimatedValue = useSharedValue(0);
const bestScoreAnimatedValue = useSharedValue(0);

const animateScoreChange = (newScore: number, previousScore: number) => {
  const scoreDiff = newScore - previousScore;

  // Animate score counter
  scoreAnimatedValue.value = withSpring(newScore, {
    damping: 15,
    stiffness: 300,
  });

  // Special animation for new best score
  if (newScore > bestScore) {
    bestScoreAnimatedValue.value = withSequence(withSpring(1.1, { damping: 15 }), withSpring(1.0, { damping: 15 }));
  }
};
```
