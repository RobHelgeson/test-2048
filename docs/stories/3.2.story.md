import { Platform, Dimensions } from 'react-native';

const { width, height } = Dimensions.get('window');

// Responsive board sizing calculation
const calculateBoardSize = () => {
  const minSize = 320;
  const maxSize = 500;
  const screenWidth = width;
  const availableSpace = screenWidth - 32 * 2; // Account for margins

  return Math.min(Math.max(availableSpace, minSize), maxSize);
};

// Platform-specific touch target sizes
const TOUCH_TARGETS = {
  ios: 44, // 44x44pt minimum
  android: 48, // 48x48dp minimum
  web: 32, // 32x32px minimum
